  10 |     | /**
  11 |     |  * @dev Contract module that allows children to implement role-based access
  12 |     |  * control mechanisms. This is a lightweight version that doesn't allow enumerating role
  13 |     |  * members except through off-chain means by accessing the contract event logs. Some
  14 |     |  * applications may benefit from on-chain enumerability, for those cases see
  15 |     |  * {AccessControlEnumerable}.
  16 |     |  *
  17 |     |  * Roles are referred to by their `bytes32` identifier. These should be exposed
  18 |     |  * in the external API and be unique. The best way to achieve this is by
  19 |     |  * using `public constant` hash digests:
  20 |     |  *
  21 |     |  * ```solidity
  22 |     |  * bytes32 public constant MY_ROLE = keccak256("MY_ROLE");
  23 |     |  * ```
  24 |     |  *
  25 |     |  * Roles can be used to represent a set of permissions. To restrict access to a
  26 |     |  * function call, use {hasRole}:
  27 |     |  *
  28 |     |  * ```solidity
  29 |     |  * function foo() public {
  30 |     |  *     require(hasRole(MY_ROLE, msg.sender));
  31 |     |  *     ...
  32 |     |  * }
  33 |     |  * ```
  34 |     |  *
  35 |     |  * Roles can be granted and revoked dynamically via the {grantRole} and
  36 |     |  * {revokeRole} functions. Each role has an associated admin role, and only
  37 |     |  * accounts that have a role's admin role can call {grantRole} and {revokeRole}.
  38 |     |  *
  39 |     |  * By default, the admin role for all roles is `DEFAULT_ADMIN_ROLE`, which means
  40 |     |  * that only accounts with this role will be able to grant or revoke other
  41 |     |  * roles. More complex role relationships can be created by using
  42 |     |  * {_setRoleAdmin}.
  43 |     |  *
  44 |     |  * WARNING: The `DEFAULT_ADMIN_ROLE` is also its own admin: it has permission to
  45 |     |  * grant and revoke this role. Extra precautions should be taken to secure
  46 |     |  * accounts that have been granted it. We recommend using {AccessControlDefaultAdminRules}
  47 |     |  * to enforce additional security measures for this role.
  48 |     |  */

  50 |     |     struct RoleData {

 184 |     |         if (!hasRole(role, account)) {
 185 |     |             _roles[role].hasRole[account] = true;
 186 |     |             emit RoleGranted(role, account, _msgSender());
 187 |     |             return true;
 188 |     |         } else {
 189 |     |             return false;
 190 |     |         }
 191 |     |     }

 197 |     |      *

  52 |     |                 CustomRevert.bubbleUpAndRevertWith(to, bytes4(0), NativeTransferFailed.selector);
  53 |     |             }
  54 |     |         } else {
  55 |     |             assembly ("memory-safe") {
  56 |     |                 // Get a pointer to some free memory.
  57 |     |                 let fmp := mload(0x40)
  58 |     | 
  59 |     |                 // Write the abi-encoded calldata into memory, beginning with the function selector.
  60 |     |                 mstore(fmp, 0xa9059cbb00000000000000000000000000000000000000000000000000000000)
  61 |     |                 mstore(add(fmp, 4), and(to, 0xffffffffffffffffffffffffffffffffffffffff)) // Append and mask the "to" argument.
  62 |     |                 mstore(add(fmp, 36), amount) // Append the "amount" argument. Masking not required as it's a full 32 byte type.
  63 |     | 
  64 |     |                 success :=
  65 |     |                     and(
  66 |     |                         // Set success to whether the call reverted, if not we check it either
  67 |     |                         // returned exactly 1 (can't just be non-zero data), or had no return data.
  68 |     |                         or(and(eq(mload(0), 1), gt(returndatasize(), 31)), iszero(returndatasize())),

  70 |     |                         // We use 0 and 32 to copy up to 32 bytes of return data into the scratch space.
  71 |     |                         // Counterintuitively, this call must be positioned second to the or() call in the
  72 |     |                         // surrounding and() call or else returndatasize() will be zero during the computation.
  73 |     |                         call(gas(), currency, 0, fmp, 68, 0, 32)
  74 |     |                     )
  75 |     | 
  76 |     |                 // Now clean the memory we used
  77 |     |                 mstore(fmp, 0) // 4 byte `selector` and 28 bytes of `to` were stored here
  78 |     |                 mstore(add(fmp, 0x20), 0) // 4 bytes of `to` and 28 bytes of `amount` were stored here
  79 |     |                 mstore(add(fmp, 0x40), 0) // 4 bytes of `amount` were stored here
  80 |     |             }
  81 |     |             // revert with ERC20TransferFailed, containing the bubbled up error as an argument
  82 |     |             if (!success) {
  83 |     |                 CustomRevert.bubbleUpAndRevertWith(
  84 |     |                     Currency.unwrap(currency), IERC20Minimal.transfer.selector, ERC20TransferFailed.selector
  85 |     |                 );
  86 |     |             }
